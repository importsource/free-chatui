⚡ Quick Start

1

Include the Script

Add the chat widget script to your HTML page

2

Configure

Set up your N8N webhook URL and customize the appearance

3

Initialize

Start the chat widget with one line of code

<!-- Include the chat widget -->
<script src="https://cdn.jsdelivr.net/gh/importsource/free-chatui@v1.0.0-alpha/n8n-chat.js"></script>

<script>
// Configure the chat widget
const N8N_CHAT_CONFIG = {
  webhookUrl: 'https://your-n8n-webhook-url',
  placeholder: 'Type a message...',
  title: 'Chat Support',
  theme: {
    primaryColor: '#007bff',
    backgroundColor: '#ffffff'
  },
  voice: {
    enabled: true
  },
  fileUpload: {
    enabled: true,
    uploadUrl: 'https://your-upload-endpoint',
    downloadUrl: 'https://your-download-endpoint'
  }
};

// Initialize the chat widget
const chatWidget = new N8NChatUI(N8N_CHAT_CONFIG);
chatWidget.init();
</script>
